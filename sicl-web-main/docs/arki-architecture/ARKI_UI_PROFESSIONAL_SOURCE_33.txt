/**
 * ARKI — Visor 3D Profesional
 * Render foto-realista con:
 * - PBR materials (metalness/roughness)
 * - HDR environment lighting
 * - Soft shadows (PCFSoft)
 * - SSAO / Ambient Occlusion
 * - Bloom / Tone mapping (ACESFilmic)
 * - Sky + sun dinámicos
 * - Post-processing (EffectComposer)
 * - Antialiasing MSAA + SMAA
 * - Cámara cinematográfica con transiciones
 * - Resoluciones hasta 16K
 */

class ViewerPro {
  constructor(container, opciones = {}) {
    this.container = container;
    this.resolucion = opciones.resolucion || '2k';
    this.calidad = opciones.calidad || 'alta'; // baja, media, alta, ultra
    this.fps = 60;

    this.escena = null;
    this.camara = null;
    this.renderer = null;
    this.controles = null;
    this.composer = null;
    this.passes = {};
    this.grupoActual = null;
    this.animaciones = [];
    this._ultimoFrame = performance.now();
    this._frames = 0;

    this._init();
  }

  static RESOLUCIONES = {
    hd: { ancho: 1280, alto: 720, label: 'HD 720p' },
    fullhd: { ancho: 1920, alto: 1080, label: 'Full HD 1080p' },
    '2k': { ancho: 2560, alto: 1440, label: '2K QHD' },
    '4k': { ancho: 3840, alto: 2160, label: '4K UHD' },
    '8k': { ancho: 7680, alto: 4320, label: '8K UHD' },
    '16k': { ancho: 15360, alto: 8640, label: '16K' },
  };

  _init() {
    const ancho = this.container.clientWidth;
    const alto = this.container.clientHeight;

    // ═══════════════════════════════════════════
    // ESCENA
    // ═══════════════════════════════════════════
    this.escena = new THREE.Scene();

    // ═══════════════════════════════════════════
    // CÁMARA
    // ═══════════════════════════════════════════
    this.camara = new THREE.PerspectiveCamera(45, ancho / alto, 0.1, 20000);
    this.camara.position.set(200, 180, 250);

    // ═══════════════════════════════════════════
    // RENDERER
    // ═══════════════════════════════════════════
    this.renderer = new THREE.WebGLRenderer({
      antialias: true,
      alpha: false,
      powerPreference: 'high-performance',
      logarithmicDepthBuffer: true,
      preserveDrawingBuffer: true, // Para captura PNG
    });
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    this.renderer.setSize(ancho, alto);
    this.renderer.shadowMap.enabled = true;
    this.renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    this.renderer.toneMapping = THREE.ACESFilmicToneMapping;
    this.renderer.toneMappingExposure = 1.0;
    this.renderer.outputColorSpace = THREE.SRGBColorSpace;
    this.renderer.physicallyCorrectLights = true;
    this.container.appendChild(this.renderer.domElement);

    // ═══════════════════════════════════════════
    // CONTROLES
    // ═══════════════════════════════════════════
    this.controles = new THREE.OrbitControls(this.camara, this.renderer.domElement);
    this.controles.enableDamping = true;
    this.controles.dampingFactor = 0.08;
    this.controles.minDistance = 20;
    this.controles.maxDistance = 5000;
    this.controles.maxPolarAngle = Math.PI / 2.05;
    this.controles.autoRotateSpeed = 0.6;
    this.controles.enablePan = true;
    this.controles.screenSpacePanning = true;

    // ═══════════════════════════════════════════
    // ILUMINACIÓN PROFESIONAL
    // ═══════════════════════════════════════════
    this._setupIluminacion();

    // ═══════════════════════════════════════════
    // ENTORNO (HDR procedural)
    // ═══════════════════════════════════════════
    this._setupEntorno();

    // ═══════════════════════════════════════════
    // SUELO Y SKY
    // ═══════════════════════════════════════════
    this._setupSuelo();
    this._setupSky();

    // ═══════════════════════════════════════════
    // GRUPO PRINCIPAL
    // ═══════════════════════════════════════════
    this.grupoActual = new THREE.Group();
    this.escena.add(this.grupoActual);

    // ═══════════════════════════════════════════
    // POST-PROCESSING
    // ═══════════════════════════════════════════
    this._setupPostProcessing();

    // ═══════════════════════════════════════════
    // LOOP
    // ═══════════════════════════════════════════
    this._animar();

    // ═══════════════════════════════════════════
    // RESIZE
    // ═══════════════════════════════════════════
    this._resizeHandler = () => this._resize();
    window.addEventListener('resize', this._resizeHandler);
  }

  _setupIluminacion() {
    // ─── Ambiente hemisférico (cielo + tierra) ───
    const hemi = new THREE.HemisphereLight(0xcfe8ff, 0x4a4a3a, 0.6);
    hemi.position.set(0, 500, 0);
    this.escena.add(hemi);

    // ─── Sol (direccional principal) ───
    this.sol = new THREE.DirectionalLight(0xfff5e6, 2.2);
    this.sol.position.set(300, 500, 200);
    this.sol.castShadow = true;
    this.sol.shadow.mapSize.width = 4096;
    this.sol.shadow.mapSize.height = 4096;
    this.sol.shadow.camera.near = 1;
    this.sol.shadow.camera.far = 3000;
    this.sol.shadow.camera.left = -800;
    this.sol.shadow.camera.right = 800;
    this.sol.shadow.camera.top = 800;
    this.sol.shadow.camera.bottom = -800;
    this.sol.shadow.bias = -0.0001;
    this.sol.shadow.normalBias = 0.02;
    this.escena.add(this.sol);

    // ─── Relleno azulado (rebote del cielo) ───
    this.rellenoCielo = new THREE.DirectionalLight(0x8ab4f8, 0.4);
    this.rellenoCielo.position.set(-200, 300, -300);
    this.escena.add(this.rellenoCielo);

    // ─── Relleno cálido (rebote del suelo) ───
    this.rellenoSuelo = new THREE.DirectionalLight(0xffd9a0, 0.25);
    this.rellenoSuelo.position.set(-100, -100, 200);
    this.escena.add(this.rellenoSuelo);
  }

  _setupEntorno() {
    // PMREM Generator para ambiente HDRI procedural
    const pmrem = new THREE.PMREMGenerator(this.renderer);
    pmrem.compileEquirectangularShader();

    // Crear sky procedural con gradiente
    const skyGeo = new THREE.SphereGeometry(5000, 64, 32);
    const skyMat = new THREE.ShaderMaterial({
      side: THREE.BackSide,
      uniforms: {
        topColor: { value: new THREE.Color(0x0a1628) },
        horizonColor: { value: new THREE.Color(0xf5e6d3) },
        bottomColor: { value: new THREE.Color(0x1a1a24) },
        offset: { value: 400 },
        exponent: { value: 0.6 },
      },
      vertexShader: `
        varying vec3 vWorldPosition;
        void main() {
          vec4 worldPosition = modelMatrix * vec4(position, 1.0);
          vWorldPosition = worldPosition.xyz;
          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
        }
      `,
      fragmentShader: `
        uniform vec3 topColor;
        uniform vec3 horizonColor;
        uniform vec3 bottomColor;
        uniform float offset;
        uniform float exponent;
        varying vec3 vWorldPosition;
        void main() {
          float h = normalize(vWorldPosition + offset).y;
          vec3 color;
          if (h > 0.0) {
            color = mix(horizonColor, topColor, pow(max(h, 0.0), exponent));
          } else {
            color = mix(horizonColor, bottomColor, pow(max(-h, 0.0), exponent));
          }
          gl_FragColor = vec4(color, 1.0);
        }
      `,
    });
    this.sky = new THREE.Mesh(skyGeo, skyMat);
    this.escena.add(this.sky);

    // Environment map desde la escena
    const envScene = new THREE.Scene();
    const envGeo = new THREE.SphereGeometry(100, 32, 16);
    const envMat = new THREE.MeshBasicMaterial({
      color: 0x8899cc,
      side: THREE.BackSide,
    });
    envScene.add(new THREE.Mesh(envGeo, envMat));

    const envMap = pmrem.fromScene(envScene, 0.04).texture;
    this.escena.environment = envMap;
    this.escena.environmentIntensity = 0.6;
  }

  _setupSuelo() {
    // Suelo mate para recibir sombras
    const geo = new THREE.PlaneGeometry(10000, 10000);
    const mat = new THREE.MeshStandardMaterial({
      color: 0x1e1e26,
      roughness: 0.95,
      metalness: 0.0,
    });
    const suelo = new THREE.Mesh(geo, mat);
    suelo.rotation.x = -Math.PI / 2;
    suelo.position.y = -0.1;
    suelo.receiveShadow = true;
    suelo.name = 'suelo';
    this.escena.add(suelo);
  }

  _setupSky() {
    // Grid sutil
    const grid = new THREE.GridHelper(4000, 80, 0x2a2a35, 0x1a1a22);
    grid.material.transparent = true;
    grid.material.opacity = 0.4;
    grid.position.y = -0.05;
    this.escena.add(grid);
    this.grid = grid;
  }

  _setupPostProcessing() {
    // Sin EffectComposer por defecto para máximo rendimiento
    // Se activa bajo demanda con activarBloom() o activarSSAO()
  }

  _animar = () => {
    requestAnimationFrame(this._animar);

    const ahora = performance.now();
    this._frames++;
    if (ahora - this._ultimoFrame >= 1000) {
      this.fps = this._frames;
      this._frames = 0;
      this._ultimoFrame = ahora;
    }

    this.controles.update();
    this.renderer.render(this.escena, this.camara);
  };

  _resize = () => {
    const ancho = this.container.clientWidth;
    const alto = this.container.clientHeight;
    if (ancho === 0 || alto === 0) return;

    this.camara.aspect = ancho / alto;
    this.camara.updateProjectionMatrix();
    this.renderer.setSize(ancho, alto);
  };

  // ═══════════════════════════════════════════════
  // RENDERIZADO DE MODELOS
  // ═══════════════════════════════════════════════

  renderDesdeModelo3D(modelo) {
    this.limpiar();

    if (!modelo || !modelo.bloques) return;

    modelo.bloques.forEach(bloque => {
      const grupoBloque = new THREE.Group();

      // ─── Geometría principal ───
      const geo = new THREE.BoxGeometry(
        bloque.ancho,
        bloque.alto,
        bloque.largo,
      );

      // ─── Material PBR ───
      const color = new THREE.Color(bloque.color || '#38bdf8');
      const mat = new THREE.MeshStandardMaterial({
        color,
        roughness: 0.55,
        metalness: 0.15,
        envMapIntensity: 0.8,
      });

      const mesh = new THREE.Mesh(geo, mat);
      mesh.position.set(
        bloque.x + bloque.ancho / 2,
        bloque.alto / 2,
        bloque.y + bloque.largo / 2,
      );
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      mesh.userData = {
        tipo: bloque.tipo,
        id: bloque.id,
        area: bloque.ancho * bloque.largo,
      };
      grupoBloque.add(mesh);

      // ─── Wireframe superior sutil ───
      const wireGeo = new THREE.EdgesGeometry(geo);
      const wireMat = new THREE.LineBasicMaterial({
        color: 0xffffff,
        transparent: true,
        opacity: 0.08,
      });
      const wire = new THREE.LineSegments(wireGeo, wireMat);
      wire.position.copy(mesh.position);
      grupoBloque.add(wire);

      // ─── Línea de piso (contorno) ───
      const contornoGeo = new THREE.BufferGeometry().setFromPoints([
        new THREE.Vector3(bloque.x, 0.1, bloque.y),
        new THREE.Vector3(bloque.x + bloque.ancho, 0.1, bloque.y),
        new THREE.Vector3(bloque.x + bloque.ancho, 0.1, bloque.y + bloque.largo),
        new THREE.Vector3(bloque.x, 0.1, bloque.y + bloque.largo),
        new THREE.Vector3(bloque.x, 0.1, bloque.y),
      ]);
      const contornoMat = new THREE.LineBasicMaterial({
        color,
        transparent: true,
        opacity: 0.6,
      });
      const contorno = new THREE.Line(contornoGeo, contornoMat);
      grupoBloque.add(contorno);

      this.grupoActual.add(grupoBloque);
    });

    // Ajustar cámara
    if (modelo.camara_sugerida) {
      this._animarCamara(
        modelo.camara_sugerida.posicion,
        modelo.camara_sugerida.target,
      );
    } else {
      this._ajustarCamara();
    }
  }

  _animarCamara(posicion, target, duracion = 1200) {
    const posInicial = this.camara.position.clone();
    const targetInicial = this.controles.target.clone();

    const posFinal = new THREE.Vector3(...posicion);
    const targetFinal = new THREE.Vector3(...target);

    const inicio = performance.now();

    const animar = () => {
      const ahora = performance.now();
      const t = Math.min((ahora - inicio) / duracion, 1);
      const easing = this._easeInOutCubic(t);

      this.camara.position.lerpVectors(posInicial, posFinal, easing);
      this.controles.target.lerpVectors(targetInicial, targetFinal, easing);
      this.controles.update();

      if (t < 1) requestAnimationFrame(animar);
    };

    animar();
  }

  _easeInOutCubic(t) {
    return t < 0.5
      ? 4 * t * t * t
      : 1 - Math.pow(-2 * t + 2, 3) / 2;
  }

  _ajustarCamara() {
    const box = new THREE.Box3().setFromObject(this.grupoActual);
    if (box.isEmpty()) return;

    const size = box.getSize(new THREE.Vector3());
    const center = box.getCenter(new THREE.Vector3());
    const maxDim = Math.max(size.x, size.y, size.z);

    this.controles.target.copy(center);
    this.camara.position.set(
      center.x + maxDim * 1.2,
      center.y + maxDim * 0.9,
      center.z + maxDim * 1.2,
    );
    this.controles.update();
  }

  limpiar() {
    while (this.grupoActual.children.length) {
      const child = this.grupoActual.children[0];
      this.grupoActual.remove(child);
      this._disposeRecursive(child);
    }
  }

  _disposeRecursive(obj) {
    if (obj.geometry) obj.geometry.dispose();
    if (obj.material) {
      if (Array.isArray(obj.material)) {
        obj.material.forEach(m => m.dispose());
      } else {
        obj.material.dispose();
      }
    }
    if (obj.children) {
      obj.children.forEach(c => this._disposeRecursive(c));
    }
  }

    return this.controles.autoRotate;
  }

  setVistaPerspectiva() {

  }

  setVistaFrontal() {

      [0, 20, 0],
    );
  }

  setVistaLateral() {
    this._animarCamara(

      [0, 20, 0],

  }

  setVistaSuperior() {

      [0, 400, 0.01],
      [0, 0, 0],
    );
  }

  setVistaIsometrica() {
    this._animarCamara(
      [250, 200, 250],
      [0, 20, 0],
    );
  }

  // ═══════════════════════════════════════════════
  // CALIDAD Y RENDER
  // ═══════════════════════════════════════════════

  setCalidad(nivel) {
    this.calidad = nivel;

    switch (nivel) {
      case 'baja':
        this.renderer.setPixelRatio(0.75);
        this.renderer.shadowMap.enabled = false;
        break;
      case 'media':
        this.renderer.setPixelRatio(1);
        this.renderer.shadowMap.enabled = true;
        this.sol.shadow.mapSize.width = 1024;
        this.sol.shadow.mapSize.height = 1024;
        break;
      case 'alta':
        this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
        this.renderer.shadowMap.enabled = true;
        this.sol.shadow.mapSize.width = 2048;
        this.sol.shadow.mapSize.height = 2048;
        break;
      case 'ultra':
        this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 3));
        this.renderer.shadowMap.enabled = true;
        this.sol.shadow.mapSize.width = 4096;
        this.sol.shadow.mapSize.height = 4096;
        this.sol.shadow.bias = -0.00005;
        break;
    }

    this.sol.shadow.map?.dispose();
    this.sol.shadow.map = null;
  }

  async capturarImagen(resolucion = '2k', opciones = {}) {
    const config = ViewerPro.RESOLUCIONES[resolucion];
    if (!config) throw new Error(`Resolución inválida: ${resolucion}`);

    // Guardar estado
    const anchoOrig = this.container.clientWidth;
    const altoOrig = this.container.clientHeight;
    const pixelRatioOrig = this.renderer.getPixelRatio();
    const aspectOrig = this.camara.aspect;

    // Ajustar renderer temporalmente
    this.renderer.setPixelRatio(1);
    this.renderer.setSize(config.ancho, config.alto, false);
    this.camara.aspect = config.ancho / config.alto;
    this.camara.updateProjectionMatrix();

    // Esperar un frame
    await new Promise(resolve => requestAnimationFrame(resolve));

    // Renderizar
    this.renderer.render(this.escena, this.camara);

    // Capturar
    const dataURL = this.renderer.domElement.toDataURL('image/png', 1.0);

    // Restaurar
    this.renderer.setPixelRatio(pixelRatioOrig);
    this.renderer.setSize(anchoOrig, altoOrig);
    this.camara.aspect = aspectOrig;
    this.camara.updateProjectionMatrix();

    return {
      dataURL,
      ancho: config.ancho,
      alto: config.alto,
      label: config.label,
      peso: this._estimarPeso(dataURL),
    };
  }

  async descargarImagen(resolucion = '2k', nombre = null) {
    const captura = await this.capturarImagen(resolucion);
    const filename = nombre || `arki_${resolucion}_${Date.now()}.png`;

    const a = document.createElement('a');
    a.href = captura.dataURL;
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);

    return {
      archivo: filename,
      resolucion: captura.label,
      peso: captura.peso,
    };
  }

  async descargarTodasLasResoluciones(nombreBase = 'render') {
    const resoluciones = ['hd', 'fullhd', '2k', '4k', '8k', '16k'];
    const resultados = [];

    for (const r of resoluciones) {
      try {
        const resultado = await this.descargarImagen(r, `${nombreBase}_${r}.png`);
        resultados.push(resultado);
        // Esperar entre descargas para no saturar
        await new Promise(resolve => setTimeout(resolve, 500));
      } catch (e) {
        console.error(`Error descargando ${r}:`, e);
      }
    }

    return resultados;
  }

  _estimarPeso(dataURL) {
    const base64 = dataURL.split(',')[1];
    const bytes = base64.length * 0.75;
    if (bytes < 1024) return `${bytes.toFixed(0)} B`;
    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
  }

  // ═══════════════════════════════════════════════
  // MODO FOTO-REALISTA
  // ═══════════════════════════════════════════════

  activarModoFotoRealista() {
    // Aumentar exposición
    this.renderer.toneMappingExposure = 1.15;

    // Aumentar environment intensity
    this.escena.environmentIntensity = 1.0;

    // Ajustar materiales existentes
    this.grupoActual.traverse(child => {
      if (child.isMesh && child.material) {
        if (Array.isArray(child.material)) {
          child.material.forEach(m => {
            m.roughness = Math.max(0.3, m.roughness - 0.2);
            m.metalness = Math.min(0.4, m.metalness + 0.1);
            m.envMapIntensity = 1.2;
            m.needsUpdate = true;
          });
        } else {
          child.material.roughness = Math.max(0.3, child.material.roughness - 0.2);
          child.material.metalness = Math.min(0.4, child.material.metalness + 0.1);
          child.material.envMapIntensity = 1.2;
          child.material.needsUpdate = true;
        }
      }
    });

    // Aumentar calidad de sombras
    this.sol.intensity = 2.8;
    this.sol.shadow.mapSize.width = 4096;
    this.sol.shadow.mapSize.height = 4096;
    this.sol.shadow.map?.dispose();
    this.sol.shadow.map = null;

    // Suelo más oscuro para contraste
    const suelo = this.escena.getObjectByName('suelo');
    if (suelo) suelo.material.color.setHex(0x0f0f14);

    return true;
  }

  desactivarModoFotoRealista() {
    this.renderer.toneMappingExposure = 1.0;
    this.escena.environmentIntensity = 0.6;
    this.sol.intensity = 2.2;

    this.grupoActual.traverse(child => {
      if (child.isMesh && child.material) {
        if (Array.isArray(child.material)) {
          child.material.forEach(m => {
            m.roughness = 0.55;
            m.metalness = 0.15;
            m.envMapIntensity = 0.8;
            m.needsUpdate = true;
          });
        } else {
          child.material.roughness = 0.55;
          child.material.metalness = 0.15;
          child.material.envMapIntensity = 0.8;
          child.material.needsUpdate = true;
        }
      }
    });

    const suelo = this.escena.getObjectByName('suelo');
    if (suelo) suelo.material.color.setHex(0x1e1e26);
  }

  // ═══════════════════════════════════════════════
  // CURSORES COLABORATIVOS
  // ═══════════════════════════════════════════════

  initCursoresColaborativos() {
    this.cursoresGrupo = new THREE.Group();
    this.escena.add(this.cursoresGrupo);
    this._cursoresMesh = new Map();
  }

  actualizarCursor(usuarioId, posicion, color = '#38bdf8', nombre = '') {
    if (!this.cursoresGrupo) this.initCursoresColaborativos();

    if (!posicion) {
      const mesh = this._cursoresMesh.get(usuarioId);
      if (mesh) {
        this.cursoresGrupo.remove(mesh);
        this._disposeRecursive(mesh);
        this._cursoresMesh.delete(usuarioId);
      }
      return;
    }

    let mesh = this._cursoresMesh.get(usuarioId);

    if (!mesh) {
      mesh = new THREE.Group();

      // Cono apuntando hacia abajo
      const geo = new THREE.ConeGeometry(1.5, 4, 8);
      const mat = new THREE.MeshBasicMaterial({ color });
      const cono = new THREE.Mesh(geo, mat);
      cono.rotation.x = Math.PI;

"""
Servicio de persistencia de versiones.
"""
import hashlib
import json
from typing import Dict, Any, List, Optional
from datetime import datetime, timedelta
from uuid import uuid4
from sqlalchemy import func
from arki.persistence.db import SessionLocal
from arki.persistence.models_extra import (
    VersionPersistente, CursorHistorico, EventoColaborativo,
)


class ServicioVersiones:

    # ═══════════════════════════════════════════
    # VERSIONES
    # ═══════════════════════════════════════════

    def crear_version(self, proyecto_id: str, params: Dict,
                       metricas: Dict = None, alertas: List = None,
                       familia_id: str = None, variante_id: str = None,
                       contexto: Dict = None, etiqueta: str = None,
                       autor_id: str = None, autor_nombre: str = None,
                       geometria_resumen: Dict = None) -> Dict:
        """Crea una nueva versión persistente."""
        db = SessionLocal()
        try:
            # Calcular número siguiente
            ultimo = db.query(func.max(VersionPersistente.numero)).filter_by(
                proyecto_id=proyecto_id
            ).scalar() or 0
            numero = ultimo + 1

            # Hash de contenido
            contenido = json.dumps({
                "params": params,
                "metricas": metricas or {},
            }, sort_keys=True, default=str)
            hash_contenido = hashlib.sha256(contenido.encode()).hexdigest()

            # Marcar anteriores como no actuales
            db.query(VersionPersistente).filter_by(
                proyecto_id=proyecto_id, es_actual=True
            ).update({"es_actual": False})

            version = VersionPersistente(
                id=str(uuid4()),
                proyecto_id=proyecto_id,
                numero=numero,
                etiqueta=etiqueta or f"v{numero}",
                autor_id=autor_id,
                autor_nombre=autor_nombre or "Anónimo",
                params=params,
                metricas=metricas or {},
                alertas=alertas or [],
                geometria_resumen=geometria_resumen or {},
                familia_id=familia_id,
                variante_id=variante_id,
                contexto=contexto or {},
                estado="aprobado",
                es_actual=True,
                hash_contenido=hash_contenido,
            )
            db.add(version)
            db.commit()
            db.refresh(version)
            return self._to_dict(version)
        except Exception:
            db.rollback()
            raise
        finally:
            db.close()

    def listar_versiones(self, proyecto_id: str,
                          limite: int = 100) -> List[Dict]:
        db = SessionLocal()
        try:
            versiones = db.query(VersionPersistente).filter_by(
                proyecto_id=proyecto_id
            ).order_by(VersionPersistente.numero.desc()).limit(limite).all()
            return [self._to_dict(v) for v in versiones]
        finally:
            db.close()

    def obtener_version(self, version_id: str) -> Optional[Dict]:
        db = SessionLocal()
        try:
            v = db.query(VersionPersistente).filter_by(id=version_id).first()
            return self._to_dict(v) if v else None
        finally:
            db.close()

    def obtener_actual(self, proyecto_id: str) -> Optional[Dict]:
        db = SessionLocal()
        try:
            v = db.query(VersionPersistente).filter_by(
                proyecto_id=proyecto_id, es_actual=True
            ).first()
            return self._to_dict(v) if v else None
        finally:
            db.close()

    def diff_versiones(self, v1_id: str, v2_id: str) -> Dict:
        v1 = self.obtener_version(v1_id)
        v2 = self.obtener_version(v2_id)
        if not v1 or not v2:
            return {"error": "Versión no encontrada"}

        # Diff params
        params_diff = {}
        all_keys = set(v1["params"].keys()) | set(v2["params"].keys())
        for k in all_keys:
            a = v1["params"].get(k)
            b = v2["params"].get(k)
            if a != b:
                params_diff[k] = {"antes": a, "despues": b}

        # Diff metricas
        metricas_diff = {}
        m1 = v1.get("metricas") or {}
        m2 = v2.get("metricas") or {}
        all_m = set(m1.keys()) | set(m2.keys())
        for k in all_m:
            a = m1.get(k)
            b = m2.get(k)
            if a != b and isinstance(a, (int, float)) and isinstance(b, (int, float)):
                metricas_diff[k] = {
                    "antes": a, "despues": b,
                    "delta": b - a,
                    "delta_pct": ((b - a) / a * 100) if a else None,
                }

        return {
            "v1": {"id": v1["id"], "numero": v1["numero"], "etiqueta": v1["etiqueta"]},
            "v2": {"id": v2["id"], "numero": v2["numero"], "etiqueta": v2["etiqueta"]},
            "params_diff": params_diff,
            "metricas_diff": metricas_diff,
            "resumen": {
                "params_cambiados": len(params_diff),
                "metricas_cambiadas": len(metricas_diff),
            },
        }

    def restaurar_version(self, version_id: str,
                           autor_id: str = None,
                           autor_nombre: str = None) -> Dict:
        """Restaura una versión creando una nueva basada en ella."""
        original = self.obtener_version(version_id)
        if not original:
            return {"error": "Versión no encontrada"}

        return self.crear_version(
            proyecto_id=original["proyecto_id"],
            params=original["params"],
            metricas=original["metricas"],
            alertas=original["alertas"],
            familia_id=original["familia_id"],
            variante_id=original["variante_id"],
            contexto=original["contexto"],
            etiqueta=f"restauración de v{original['numero']}",
            autor_id=autor_id,
            autor_nombre=autor_nombre,
        )

    def _to_dict(self, v: VersionPersistente) -> Dict:
        return {
            "id": v.id,
            "proyecto_id": v.proyecto_id,
            "numero": v.numero,
            "etiqueta": v.etiqueta,
            "autor_id": v.autor_id,
            "autor_nombre": v.autor_nombre,
            "params": v.params,
            "metricas": v.metricas,
            "alertas": v.alertas,
            "geometria_resumen": v.geometria_resumen,
            "familia_id": v.familia_id,
            "variante_id": v.variante_id,
            "contexto": v.contexto,
            "estado": v.estado,
            "es_actual": v.es_actual,
            "hash_contenido": v.hash_contenido,
            "creado": v.creado.isoformat() if v.creado else None,
        }

    # ═══════════════════════════════════════════
    # CURSORES
    # ═══════════════════════════════════════════

    def registrar_cursor(self, proyecto_id: str, usuario_id: str,
                          usuario_nombre: str, usuario_color: str,
                          posicion: Dict[str, float],
                          vista: str = "3d") -> str:
        """Registra un cursor para replay histórico."""
        db = SessionLocal()
        try:
            cursor = CursorHistorico(
                id=str(uuid4()),
                proyecto_id=proyecto_id,
                usuario_id=usuario_id,
                usuario_nombre=usuario_nombre,
                usuario_color=usuario_color,
                x=posicion.get("x", 0),
                y=posicion.get("y", 0),
                z=posicion.get("z", 0),
                vista=vista,
            )
            db.add(cursor)
            db.commit()
            return cursor.id
        finally:
            db.close()

    def obtener_cursores(self, proyecto_id: str,

                CursorHistorico.timestamp >= desde,

"""
Mapeador IFC geométrico detallado por familia.
Genera geometría IFC 4X3 con B-rep, extrusión y representaciones específicas.
"""
from typing import Dict, Any, List
from datetime import datetime
import hashlib


class MapeadorIFCGeometrico:
    """
    Genera IFC detallado con:
    - Representaciones geométricas específicas por familia
    - Materiales y propiedades
    - Sistemas (HVAC, eléctrico, sanitario)
    - Estructura espacial completa
    """

    def __init__(self, nombre_proyecto: str = "ARKI_Project"):
        self.nombre = nombre_proyecto
        self._id = 0
        self.entidades: List[str] = []
        self.guid_map: Dict[str, str] = {}
        self._materiales: Dict[str, int] = {}

    def _guid(self, key: str) -> str:
        if key in self.guid_map:
            return self.guid_map[key]
        h = hashlib.md5(f"{self.nombre}_{key}".encode()).hexdigest().upper()
        guid = f"{h[:8]}-{h[8:12]}-{h[12:16]}-{h[16:20]}-{h[20:32]}"
        self.guid_map[key] = guid
        return guid

    def _add(self, tipo: str, attrs: List) -> int:
        self._id += 1
        attr_str = ",".join(self._fmt(a) for a in attrs)
        self.entidades.append(f"#{self._id}={tipo}({attr_str});")
        return self._id

    def _fmt(self, v):
        if v is None: return "$"
        if isinstance(v, bool): return ".T." if v else ".F."
        if isinstance(v, (int, float)): return str(v)
        if isinstance(v, list): return f"({','.join(self._fmt(x) for x in v)})"
        if isinstance(v, str) and v.startswith("#"): return v
        return f"'{v}'"

    def _ref(self, eid): return f"#{eid}"

    # ═══════════════════════════════════════════
    # EXPORTAR POR FAMILIA
    # ═══════════════════════════════════════════

    def exportar(self, familia_id: str, datos: Dict[str, Any]) -> str:
        header = self._header(f"{familia_id}_{self.nombre}")
        self._unidades()
        self._contexto_geometrico()

        # Jerarquía espacial
        proyecto = self._crear_proyecto(datos)
        sitio = self._crear_sitio(proyecto, datos)
        edificio = self._crear_edificio(sitio, datos)

        # Pisos
        pisos = self._crear_pisos(edificio, datos)

        # Materiales por familia
        materiales = self._crear_materiales_familia(familia_id)
        self._materiales = materiales

        # Geometría específica por familia
        if familia_id == "hospital":
            self._geometria_hospital(datos, pisos)
        elif familia_id == "educacion":
            self._geometria_educacion(datos, pisos)
        elif familia_id == "comercio":
            self._geometria_comercio(datos, pisos)
        elif familia_id == "oficinas":
            self._geometria_oficinas(datos, pisos)
        elif familia_id == "transporte":
            self._geometria_transporte(datos, pisos)
        elif familia_id == "deporte":
            self._geometria_deporte(datos, pisos)
        elif familia_id == "paisajismo":
            self._geometria_paisajismo(datos, pisos)
        elif familia_id == "industria":
            self._geometria_industria(datos, pisos)
        elif familia_id == "residencial":
            self._geometria_residencial(datos, pisos)
        elif familia_id == "hospedaje":
            self._geometria_hospedaje(datos, pisos)
        elif familia_id == "cultura":
            self._geometria_cultura(datos, pisos)
        elif familia_id == "culto":
            self._geometria_culto(datos, pisos)
        else:
            self._geometria_generica(datos, pisos)

        # Property sets
        self._crear_property_sets(datos, familia_id)

        # Sistemas (MEP)
        self._crear_sistemas(familia_id, datos)

        return header + "\n".join(self.entidades) + self._footer()

    # ═══════════════════════════════════════════
    # INFRAESTRUCTURA IFC
    # ═══════════════════════════════════════════

    def _header(self, nombre: str) -> str:
        return f"""ISO-10303-21;
HEADER;
FILE_DESCRIPTION(('ViewDefinition [CoordinationView_V2.0]','Option [AP242]'),'2;1');
FILE_NAME('{nombre}.ifc','{datetime.utcnow().isoformat()}',('ARKI'),('ARKI'),'ARKI v3.0','ARKI','');
FILE_SCHEMA(('IFC4X3'));
ENDSEC;
DATA;
"""

    def _footer(self) -> str:
        return "\nENDSEC;\nEND-ISO-10303-21;\n"

    def _unidades(self):
        self._add("IFCSIUNIT", ["*", ".LENGTHUNIT.", ".METRE.", None])
        self._add("IFCSIUNIT", ["*", ".AREAUNIT.", ".SQUARE_METRE.", None])
        self._add("IFCSIUNIT", ["*", ".VOLUMEUNIT.", ".CUBIC_METRE.", None])
        self._add("IFCSIUNIT", ["*", ".PLANEANGLEUNIT.", ".RADIAN.", None])

    def _contexto_geometrico(self):
        origin = self._add("IFCCARTESIANPOINT", [0.0, 0.0, 0.0])
        z_axis = self._add("IFCDIRECTION", [0.0, 0.0, 1.0])
        x_axis = self._add("IFCDIRECTION", [1.0, 0.0, 0.0])
        wp = self._add("IFCAXIS2PLACEMENT3D", [self._ref(origin), self._ref(z_axis), self._ref(x_axis)])
        ctx = self._add("IFCGEOMETRICREPRESENTATIONCONTEXT",
                        [None, "Model", 3, 1.0E-5, self._ref(wp), None])

    def _crear_proyecto(self, datos: Dict) -> int:
        return self._add("IFCPROJECT", [
            self._guid("project"), None,
            self._fmt(datos.get("nombre", self.nombre)),

        ])

    def _crear_sitio(self, proyecto_id: int, datos: Dict) -> int:

    def _geometria_hospital(self, datos: Dict, pisos: List[int]):
        """Geometría específica de hospital."""
        planta = datos.get("planta", {})
        elementos = planta.get("elementos", [])

        for el in elementos:
            if el.get("tipo") in ("emergencia", "consulta_externa", "hospitalizacion", "uci"):

    def _crear_espacio_hospitalario(self, el: Dict, pisos: List[int], datos: Dict):
        """Crea un IFCSPACE con propiedades hospitalarias."""
        piso_id = pisos[0] if pisos else None

        # Posicionamiento
        origin = self._add("IFCCARTESIANPOINT", [el.get("x", 0), el.get("y", 0), 0])
        z_axis = self._add("IFCDIRECTION", [0, 0, 1])
        x_axis = self._add("IFCDIRECTION", [1, 0, 0])
        axis = self._add("IFCAXIS2PLACEMENT3D", [self._ref(origin), self._ref(z_axis), self._ref(x_axis)])
        placement = self._add("IFCLOCALPLACEMENT", [self._ref(piso_id) if piso_id else None, self._ref(axis)])

        # Representación geométrica
        ancho = el.get("ancho", 30)
        largo = el.get("largo", 30)
        altura = 3.5 * datos.get("num_pisos", 5)

        # Profile rectangular
        profile = self._add("IFCRECTANGLEPROFILEDEF", [
            ".AREA.", None, ancho, largo,
        ])
        dir_z = self._add("IFCDIRECTION", [0, 0, 1])
        solid = self._add("IFCEXTRUDEDAREASOLID", [
            self._ref(profile), self._ref(axis), self._ref(dir_z), altura,
        ])

        # Shape representation
        shape = self._add("IFCSHAPEREPRESENTATION", [
            self._guid("context"), "Body", "SweptSolid",
            [self._ref(solid)],
        ])
        prod_def = self._add("IFCPRODUCTDEFINITIONSHAPE", [
            None, None, [self._ref(shape)],
        ])

        # Space
        space = self._add("IFCSPACE", [
            self._guid(f"space_{el.get('id', 'x')}"), None,
            self._fmt(el.get("nombre", "Espacio")),
            None, None,
            self._ref(placement),
            self._ref(prod_def),
            None,
            ".ELEMENT.",
            None,
            None,
        ])

        for el in elementos:
            piso_idx = 0
            if el.get("tipo") == "aulas":

            piso_id = pisos[piso_idx]

            origin = self._add("IFCCARTESIANPOINT", [el.get("x", 0), el.get("y", 0), 0])
            axis = self._add("IFCAXIS2PLACEMENT3D", [self._ref(origin), None, None])
            placement = self._add("IFCLOCALPLACEMENT", [self._ref(piso_id), self._ref(axis)])

            self._add("IFCSPACE", [
                self._guid(f"edu_{el.get('id', 'x')}"), None,
                self._fmt(el.get("nombre", "Aula")),
                None, None, self._ref(placement), None, None,
                ".ELEMENT.", None, None,

from fastapi import APIRouter, Depends, HTTPException
from fastapi.responses import FileResponse
from pydantic import BaseModel
from typing import Dict, Any
import tempfile

from arki.auth.dependencies import usuario_actual
from arki.ifc.geometry_mapper import exportar_ifc_geometrico_detallado


router = APIRouter(prefix="/ifc-geometrico", tags=["ifc-geometrico"])


class ExportarInput(BaseModel):
    familia_id: str
    datos: Dict[str, Any]


@router.post("/exportar")
def exportar_ifc(data: ExportarInput,
                  usuario: dict = Depends(usuario_actual)):
    """Exporta IFC 4X3 geométricamente detallado por familia."""
    with tempfile.NamedTemporaryFile(suffix=".ifc", delete=False) as f:
        ruta = f.name

    exportar_ifc_geometrico_detallado(data.familia_id, data.datos, ruta)

    return FileResponse(
        ruta,
        media_type="application/x-step",
        filename=f"{data.familia_id}_{data.datos.get('nombre', 'proyecto')}.ifc",
    )
