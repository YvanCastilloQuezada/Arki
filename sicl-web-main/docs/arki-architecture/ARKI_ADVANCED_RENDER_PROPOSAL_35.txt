/**
 * ARKI — Path Tracer
 * Render foto-realista por acumulación de muestras.
 * Usa three-gpu-pathtracer.
 */

import { WebGLPathTracer } from 'https://esm.sh/three-gpu-pathtracer@0.0.23';

export class PathTracerARKI {
  constructor(viewer) {
    this.viewer = viewer;
    this.tracer = null;
    this.muestras = 0;
    this.activo = false;
    this.maxMuestras = 500;
    this.onProgreso = null;
  }

  inicializar() {
    if (!this.viewer.renderer.capabilities.isWebGL2) {
      throw new Error('Path tracing requiere WebGL2');
    }

    this.tracer = new WebGLPathTracer(this.viewer.renderer);
    this.tracer.setScene(this.viewer.escena, this.viewer.camara);
    this.tracer.bounces = 5;
    this.tracer.renderScale = 1.0;
    this.tracer.tiles.set(3, 3);
    this.tracer.minSamples = 5;
    this.tracer.multipleImportanceSampling = true;
    this.tracer.renderToCanvas = true;

    // Filtrar geometría incompatible
    this._prepararEscena();

    return this.tracer;
  }

  _prepararEscena() {
    // Path tracer solo soporta MeshStandardMaterial y MeshPhysicalMaterial
    this.viewer.escena.traverse((obj) => {
      if (obj.isMesh && obj.material) {
        const mat = obj.material;
        if (!mat.isMeshStandardMaterial && !mat.isMeshPhysicalMaterial) {
          // Convertir a StandardMaterial
          obj.material = new THREE.MeshStandardMaterial({
            color: mat.color || 0xcccccc,
            map: mat.map || null,
            roughness: mat.roughness || 0.7,
            metalness: mat.metalness || 0.1,
          });
        }
      }
    });

    // Luz direccional requiere MIS activado
    if (this.viewer.sol) {
      this.viewer.sol.castShadow = true;
    }
  }

  activar() {
    if (!this.tracer) this.inicializar();
    this.activo = true;
    this.muestras = 0;
    this.tracer.updateCamera();
    this.tracer.updateMaterials();
    this.tracer.updateLights();
    this._loop();
  }

  desactivar() {
    this.activo = false;
    // Volver al render normal
    this.viewer._animar();
  }

  reset() {
    this.muestras = 0;
    if (this.tracer) this.tracer.reset();
  }

  _loop = () => {
    if (!this.activo) return;

    if (this.muestras < this.maxMuestras) {
      this.tracer.renderSample();
      this.muestras++;

      if (this.onProgreso) {
        this.onProgreso(this.muestras, this.maxMuestras);
      }
    }

    requestAnimationFrame(this._loop);
  };

  /**
   * Renderiza N muestras sin animar la UI (para exportar).
   */
  async renderOffline(muestras = 500) {
    for (let i = 0; i < muestras; i++) {
      this.tracer.renderSample();
      if (i % 50 === 0) {
        await new Promise(r => setTimeout(r, 0));
        if (this.onProgreso) this.onProgreso(i, muestras);
      }
    }
    return this.viewer.renderer.domElement.toDataURL('image/png');
  }

  /**
   * Ajusta la cámara y resetea el trazado.
   */
  actualizarCamara() {
    if (this.tracer) {
      this.tracer.updateCamera();
      this.muestras = 0;
    }
  }
}

    cancelAnimationFrame(this._animationId);
    this._animationId = null;
  }

  this._animar();

/**
 * ARKI — WebXR Manager
 * VR y AR desde el navegador.
 */

export class XRManager {
  constructor(viewer) {
    this.viewer = viewer;
    this.sesionActual = null;
    this.controladores = [];
    this.soporteVR = false;
    this.soporteAR = false;
  }

  async detectarSoporte() {
    if (!navigator.xr) {
      return { vr: false, ar: false, razon: 'WebXR no disponible' };
    }

    this.soporteVR = await navigator.xr.isSessionSupported('immersive-vr');
    this.soporteAR = await navigator.xr.isSessionSupported('immersive-ar');

    return {
      vr: this.soporteVR,
      ar: this.soporteAR,
    };
  }

  async entrarVR() {
    if (!this.soporteVR) throw new Error('VR no soportado');

    const session = await navigator.xr.requestSession('immersive-vr', {
      optionalFeatures: ['local-floor', 'bounded-floor', 'hand-tracking'],
    });

    await this.viewer.renderer.xr.setSession(session);
    this.viewer.renderer.xr.enabled = true;
    this.sesionActual = session;

    // Configurar controladores
    this._setupControladores();

    // Ajustar cámara al inicio
    if (session.requestReferenceSpace) {
      const refSpace = await session.requestReferenceSpace('local-floor');
      this.viewer.renderer.xr.setReferenceSpace(refSpace);
    }

    session.addEventListener('end', () => {
      this.sesionActual = null;
      this.viewer.renderer.xr.enabled = false;
    });

    return session;
  }

  async entrarAR() {
    if (!this.soporteAR) throw new Error('AR no soportado');

    const session = await navigator.xr.requestSession('immersive-ar', {
      requiredFeatures: ['hit-test'],
      optionalFeatures: ['anchors', 'light-estimation', 'dom-overlay'],
      domOverlay: { root: document.querySelector('#ar-overlay') },
    });

    await this.viewer.renderer.xr.setSession(session);
    this.viewer.renderer.xr.enabled = true;
    this.sesionActual = session;

    // Configurar hit-test
    await this._setupHitTest(session);

    session.addEventListener('end', () => {
      this.sesionActual = null;
      this.viewer.renderer.xr.enabled = false;
      this._limpiarReticula();
    });

    return session;
  }

  _setupControladores() {
    const controlador0 = this.viewer.renderer.xr.getController(0);
    const controlador1 = this.viewer.renderer.xr.getController(1);

    controlador0.addEventListener('selectstart', () => this._onSelect(0));
    controlador1.addEventListener('selectstart', () => this._onSelect(1));

    // Rayo visual
    [controlador0, controlador1].forEach(c => {
      const geo = new THREE.BufferGeometry().setFromPoints([
        new THREE.Vector3(0, 0, 0),
        new THREE.Vector3(0, 0, -5),
      ]);
      const line = new THREE.Line(geo, new THREE.LineBasicMaterial({
        color: 0xd4ff3a,
      }));
      c.add(line);
    });

    this.viewer.escena.add(controlador0, controlador1);
    this.controladores = [controlador0, controlador1];
  }

  async _setupHitTest(session) {
    const viewerSpace = await session.requestReferenceSpace('viewer');
    const hitTestSource = await session.requestHitTestSource({
      space: viewerSpace,
    });

    // Retícula
    const reticleGeo = new THREE.RingGeometry(0.15, 0.2, 32);
    reticleGeo.rotateX(-Math.PI / 2);
    const reticleMat = new THREE.MeshBasicMaterial({
      color: 0xd4ff3a,
      transparent: true,
      opacity: 0.8,
    });
    this.reticula = new THREE.Mesh(reticleGeo, reticleMat);
    this.reticula.visible = false;
    this.reticula.matrixAutoUpdate = false;
    this.viewer.escena.add(this.reticula);

    // Loop de hit-test
    this.viewer.renderer.setAnimationLoop((timestamp, frame) => {
      if (!frame) return;
      const refSpace = this.viewer.renderer.xr.getReferenceSpace();
      const hitResults = frame.getHitTestResults(hitTestSource);

      if (hitResults.length > 0) {
        const pose = hitResults[0].getPose(refSpace);
        this.reticula.visible = true;
        this.reticula.matrix.fromArray(pose.transform.matrix);
      } else {
        this.reticula.visible = false;
      }

      this.viewer.renderer.render(this.viewer.escena, this.viewer.camara);
    });
  }

  _limpiarReticula() {
    if (this.reticula) {
      this.viewer.escena.remove(this.reticula);
      this.reticula.geometry.dispose();
      this.reticula.material.dispose();
      this.reticula = null;
    }
    this.controladores.forEach(c => this.viewer.escena.remove(c));
    this.controladores = [];
  }

  _onSelect(indice) {
    if (this.sesionActual?.mode === 'immersive-ar' && this.reticula?.visible) {
      // Colocar el modelo en la posición de la retícula
      const posicion = new THREE.Vector3();
      posicion.setFromMatrixPosition(this.reticula.matrix);
      // Aquí puedes hacer un raycast o clonar el modelo
    }
  }
}

"""
Cliente del servicio de render en la nube.
"""
import httpx
import os
from typing import Dict, Any, Optional


class ClienteRenderNube:
    def __init__(self):
        self.url = os.environ.get(
            'ARKI_RENDER_URL',
            'http://render-service:9000',
        )
        self.token = os.environ.get('ARKI_RENDER_TOKEN', '')

    async def solicitar_render(
        self,
        familia_id: str,
        params: Dict[str, Any],
        resolucion: str = '4k',
        muestras: int = 200,
    ) -> bytes:
        """Solicita render remoto y devuelve PNG."""
        async with httpx.AsyncClient(timeout=300) as client:
            res = await client.post(
                f'{self.url}/render',
                json={
                    'familia_id': familia_id,
                    'params': params,
                    'resolucion': resolucion,
                    'muestras': muestras,
                },
                headers={'Authorization': f'Bearer {self.token}'},
            )
            res.raise_for_status()
            return res.content

python
"""
Servicio de render GPU dedicado.
Corre en AWS G4dn / GCP T4 / cualquier GPU.
"""
from fastapi import FastAPI, HTTPException, Header
from pydantic import BaseModel
from typing import Dict, Any
import base64
import subprocess
import tempfile
import os
import hashlib

app = FastAPI(title="ARKI Render Service")

RESOLUCIONES = {
    'hd': (1280, 720),
    'fullhd': (1920, 1080),
    '2k': (2560, 1440),
    '4k': (3840, 2160),
    '8k': (7680, 4320),
    '16k': (15360, 8640),
}


class RenderInput(BaseModel):
    familia_id: str
    params: Dict[str, Any]
    resolucion: str = '4k'
    muestras: int = 200


@app.post("/render")
async def render(data: RenderInput, authorization: str = Header(None)):
    """Renderiza usando Blender headless con GPU."""
    if data.resolucion not in RESOLUCIONES:
        raise HTTPException(400, f'Resolución inválida: {data.resolucion}')

    ancho, alto = RESOLUCIONES[data.resolucion]

    # Guardar params en archivo temporal
    with tempfile.TemporaryDirectory() as tmpdir:
        params_path = os.path.join(tmpdir, 'params.json')
        with open(params_path, 'w') as f:
            import json
            json.dump(data.params, f)

        output_path = os.path.join(tmpdir, 'render.png')

        # Invocar Blender headless
        comando = [
            'blender',
            '--background',
            '--python', '/app/render_script.py',
            '--',
            '--familia', data.familia_id,
            '--params', params_path,
            '--output', output_path,
            '--ancho', str(ancho),
            '--alto', str(alto),
            '--muestras', str(data.muestras),
        ]

        try:
            subprocess.run(comando, check=True, timeout=600)
        except subprocess.TimeoutExpired:
            raise HTTPException(500, 'Render timeout')
        except subprocess.CalledProcessError as e:
            raise HTTPException(500, f'Error Blender: {e}')

        with open(output_path, 'rb') as f:
            return f.read()


@app.get("/health")
def health():
    """Verifica que GPU está disponible."""
    try:
        result = subprocess.run(
            ['nvidia-smi', '--query-gpu=name,memory.total', '--format=csv,noheader'],
            capture_output=True, text=True, timeout=5,
        )
        gpu_info = result.stdout.strip()
        return {'ok': True, 'gpu': gpu_info}
    except Exception as e:
        return {'ok': False, 'error': str(e)}

/**
 * ARKI — Efectos cinematográficos
 * DOF + Motion Blur + Bloom + Vignette.
 */

import { EffectComposer } from 'three/addons/postprocessing/EffectComposer.js';
import { RenderPass } from 'three/addons/postprocessing/RenderPass.js';
import { BokehPass } from 'three/addons/postprocessing/BokehPass.js';
import { UnrealBloomPass } from 'three/addons/postprocessing/UnrealBloomPass.js';
import { OutputPass } from 'three/addons/postprocessing/OutputPass.js';
import { ShaderPass } from 'three/addons/postprocessing/ShaderPass.js';
import { VignetteShader } from 'three/addons/shaders/VignetteShader.js';

export class EfectosCine {
  constructor(viewer) {
    this.viewer = viewer;
    this.composer = null;
    this.passes = {};
    this.activo = false;
  }

  inicializar() {
    const { renderer, escena, camara } = this.viewer;
    const { clientWidth: w, clientHeight: h } = this.viewer.container;

    this.composer = new EffectComposer(renderer);
    this.composer.setSize(w, h);
    this.composer.setPixelRatio(Math.min(window.devicePixelRatio, 2));

    // Render base
    this.passes.render = new RenderPass(escena, camara);
    this.composer.addPass(this.passes.render);

    // DOF (bokeh)
    this.passes.bokeh = new BokehPass(escena, camara, {
      focus: 50.0,
      aperture: 0.0001,
      maxblur: 0.01,
    });
    this.composer.addPass(this.passes.bokeh);

    // Bloom sutil
    this.passes.bloom = new UnrealBloomPass(
      new THREE.Vector2(w, h),
      0.4,   // strength
      0.6,   // radius
      0.85,  // threshold
    );
    this.composer.addPass(this.passes.bloom);

    // Vignette
    this.passes.vignette = new ShaderPass(VignetteShader);
    this.passes.vignette.uniforms['offset'].value = 1.2;
    this.passes.vignette.uniforms['darkness'].value = 1.1;
    this.composer.addPass(this.passes.vignette);

    // Output final
    this.passes.output = new OutputPass();
    this.composer.addPass(this.passes.output);

    // Reemplazar el render loop
    this._parchearLoop();
  }

  _parchearLoop() {
    const self = this;
    const animarOriginal = this.viewer._animar;

    this.viewer._animar = function() {
      requestAnimationFrame(this._animar.bind(this));

      const ahora = performance.now();
      this._frames++;
      if (ahora - this._ultimoFrame >= 1000) {
        this.fps = this._frames;
        this._frames = 0;
        this._ultimoFrame = ahora;
      }

      this.controles.update();

      if (self.activo && self.composer) {
        self.composer.render();
      } else {
        this.renderer.render(this.escena, this.camara);
      }
    };
  }

  activar() {
    if (!this.composer) this.inicializar();
    this.activo = true;
  }

  desactivar() {
    this.activo = false;
  }

  setFoco(distancia) {
    if (this.passes.bokeh) {
      this.passes.bokeh.uniforms['focus'].value = distancia;
    }
  }

  setApertura(valor) {
    if (this.passes.bokeh) {
      this.passes.bokeh.uniforms['aperture'].value = valor * 0.0001;
    }
  }

  setBloom(intensidad) {
    if (this.passes.bloom) {
      this.passes.bloom.strength = intensidad;
    }
  }

  setMotionBlur(activo) {
    // Motion blur necesita shader custom con buffer de velocidad
    // Simplificado: aumentar el bokeh
    if (activo) {
      this.passes.bokeh.uniforms['maxblur'].value = 0.02;
    } else {
      this.passes.bokeh.uniforms['maxblur'].value = 0.01;
    }
  }
}

/**
 * ARKI — Grabador de video recorrido.
 * Captura animaciones de cámara como WebM/MP4.
 */

export class VideoRecorder {
  constructor(viewer) {
    this.viewer = viewer;
    this.recorder = null;
    this.chunks = [];
    this.grabando = false;
  }

  /**
   * Graba un recorrido por keyframes.
   * @param {Array} keyframes - [{posicion, target, duracion}]
   * @param {String} nombre
   * @param {String} formato - 'webm' | 'mp4'
   */
  async grabar(keyframes, nombre = 'recorrido', formato = 'webm') {
    if (this.grabando) throw new Error('Ya hay una grabación en curso');

    const canvas = this.viewer.renderer.domElement;
    const fps = 60;

    // Workaround Android: usar canvas virtual
    const virtualCanvas = document.createElement('canvas');
    virtualCanvas.width = canvas.width;
    virtualCanvas.height = canvas.height;
    const vctx = virtualCanvas.getContext('2d');

    const stream = virtualCanvas.captureStream(fps);

    // Elegir mimeType
    const mimeTypes = {
      webm: 'video/webm;codecs=vp9',
      mp4: 'video/mp4;codecs=avc1',
    };

    const mimeType = MediaRecorder.isTypeSupported(mimeTypes[formato])
      ? mimeTypes[formato]
      : 'video/webm';

    this.recorder = new MediaRecorder(stream, {
      mimeType,
      videoBitsPerSecond: 20_000_000,
    });

    this.chunks = [];
    this.recorder.ondataavailable = (e) => {
      if (e.data.size > 0) this.chunks.push(e.data);
    };

    const promesaFin = new Promise((resolve) => {
      this.recorder.onstop = () => resolve();
    });

    this.recorder.start();
    this.grabando = true;

    // Animar por keyframes
    const duracionTotal = keyframes.reduce(
      (acc, kf) => acc + (kf.duracion || 3000),
      0,
    );

    const inicio = performance.now();

    await new Promise((resolve) => {
      const animar = () => {
        const t = (performance.now() - inicio) / duracionTotal;
        if (t >= 1) {
          resolve();
          return;
        }

        // Encontrar keyframe actual
        let acumulado = 0;
        let keyframeActual = null;
        let keyframeSiguiente = null;
        let tLocal = 0;

        for (let i = 0; i < keyframes.length; i++) {
          const kf = keyframes[i];
          const dur = (kf.duracion || 3000) / duracionTotal;

          if (t >= acumulado && t < acumulado + dur) {
            keyframeActual = kf;
            keyframeSiguiente = keyframes[i + 1] || kf;
            tLocal = (t - acumulado) / dur;
            break;
          }
          acumulado += dur;
        }

        if (keyframeActual && keyframeSiguiente) {
          this._interpolarCamara(
            keyframeActual,
            keyframeSiguiente,
            tLocal,
          );
        }

        // Render y copiar
        this.viewer.renderer.render(
          this.viewer.escena,
          this.viewer.camara,
        );
        vctx.drawImage(canvas, 0, 0);

        requestAnimationFrame(animar);
      };

      animar();
    });

    // Detener
    this.recorder.stop();
    await promesaFin;
    this.grabando = false;

    // Descargar
    const blob = new Blob(this.chunks, { type: mimeType });
    this._descargar(blob, `${nombre}_${Date.now()}.${formato}`);

    return blob;
  }

  _interpolarCamara(kfA, kfB, t) {
    const easing = this._easeInOutCubic(t);

    const posA = new THREE.Vector3(...kfA.posicion);
    const posB = new THREE.Vector3(...kfB.posicion);
    const targetA = new THREE.Vector3(...(kfA.target || [0, 0, 0]));
    const targetB = new THREE.Vector3(...(kfB.target || [0, 0, 0]));

    this.viewer.camara.position.lerpVectors(posA, posB, easing);
    this.viewer.controles.target.lerpVectors(targetA, targetB, easing);
    this.viewer.controles.update();
  }

  _easeInOutCubic(t) {
    return t < 0.5
      ? 4 * t * t * t
      : 1 - Math.pow(-2 * t + 2, 3) / 2;
  }

  _descargar(blob, nombre) {
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = nombre;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  }

  /**
   * Recorrido preestablecido: 360° alrededor del modelo.
   */
  async grabarRecorridoCircular(duracion = 15000, altura = 150) {
    const box = new THREE.Box3().setFromObject(this.viewer.grupoActual);
    const center = box.getCenter(new THREE.Vector3());
    const size = box.getSize(new THREE.Vector3());
    const radio = Math.max(size.x, size.z) * 1.8;

    const keyframes = [];
    const numPasos = 8;

    for (let i = 0; i <= numPasos; i++) {
      const angulo = (i / numPasos) * Math.PI * 2;
      keyframes.push({
        posicion: [
          center.x + Math.cos(angulo) * radio,
          center.y + altura,
          center.z + Math.sin(angulo) * radio,
        ],
        target: [center.x, center.y, center.z],
        duracion: duracion / numPasos,
      });
    }

    return this.grabar(keyframes, 'recorrido_360', 'webm');
  }

  /**
   * Recorrido de aproximación (dolly in).
   */
  async grabarDollyIn(duracion = 8000) {
    const box = new THREE.Box3().setFromObject(this.viewer.grupoActual);
    const center = box.getCenter(new THREE.Vector3());

    const keyframes = [
      {
        posicion: [center.x + 400, center.y + 300, center.z + 400],
        target: [center.x, center.y, center.z],
        duracion: duracion * 0.6,
      },
      {
        posicion: [center.x + 100, center.y + 50, center.z + 100],
        target: [center.x, center.y + 20, center.z],
        duracion: duracion * 0.4,
      },
    ];

    return this.grabar(keyframes, 'recorrido_dolly', 'webm');
  }
}

import { USDZExporter } from 'three/addons/exporters/USDZExporter.js';

export class USDZExporterARKI {
  constructor(viewer) {
    this.viewer = viewer;
  }

  async exportar(nombre = 'modelo') {
    // Preparar escena: solo geometría visible
    const escenaExport = new THREE.Scene();

    // Luces básicas para USDZ
    const luzAmbiente = new THREE.AmbientLight(0xffffff, 0.6);
    escenaExport.add(luzAmbiente);

    const luzDir = new THREE.DirectionalLight(0xffffff, 1.0);
    luzDir.position.set(1, 2, 1);
    escenaExport.add(luzDir);

    // Copiar geometría
    const grupoClon = this.viewer.grupoActual.clone(true);
    grupoClon.traverse((obj) => {
      if (obj.isMesh && obj.material) {
        // USDZ soporta MeshStandardMaterial
        if (!obj.material.isMeshStandardMaterial) {
          obj.material = new THREE.MeshStandardMaterial({
            color: obj.material.color || 0xcccccc,
            roughness: 0.7,
            metalness: 0.1,
          });
        }
      }
    });
    escenaExport.add(grupoClon);

    // Exportar
    const exporter = new USDZExporter();
    const arrayBuffer = await exporter.parseAsync(escenaExport, {
      quickLookCompatible: true,
      includeAnchoringProperties: true,
      maxTextureSize: 2048,
    });

    const blob = new Blob([arrayBuffer], {
      type: 'model/vnd.usdz+zip',
    });

    this._descargar(blob, `${nombre}_${Date.now()}.usdz`);
    return blob;
  }

  _descargar(blob, nombre) {
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = nombre;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);

/**
 * ARKI — Comparador con slider visual
 * Muestra dos versiones del modelo con un slider.
 */

export class SliderComparador {
  constructor(container, viewerA, viewerB) {
    this.container = container;
    this.viewerA = viewerA;
    this.viewerB = viewerB;
    this.slider = 0.5;
    this._init();
  }

  _init() {
    // Panel de dos capas superpuestas
    this.container.innerHTML = `
      <div class="slider-comparador">
        <div class="slider-canvas-a" id="canvasA"></div>
        <div class="slider-canvas-b" id="canvasB"></div>
        <div class="slider-line" id="sliderLine"></div>
        <div class="slider-handle" id="sliderHandle">
          <div class="slider-handle-line"></div>
        </div>
        <div class="slider-labels">
          <span class="slider-label-a" id="labelA">Versión A</span>
          <span class="slider-label-b" id="labelB">Versión B</span>
        </div>
      </div>
    `;

    this.canvasA = this.container.querySelector('#canvasA');
    this.canvasB = this.container.querySelector('#canvasB');
    this.sliderLine = this.container.querySelector('#sliderLine');
    this.sliderHandle = this.container.querySelector('#sliderHandle');

    this._bindEventos();
    this.setSlider(0.5);
  }

  _bindEventos() {
    let arrastrando = false;

    const iniciar = () => { arrastrando = true; };
    const terminar = () => { arrastrando = false; };

    const mover = (e) => {
      if (!arrastrando) return;
      const rect = this.container.getBoundingClientRect();
      const x = (e.clientX || e.touches?.[0]?.clientX) - rect.left;
      const pct = Math.max(0, Math.min(1, x / rect.width));
      this.setSlider(pct);
    };

    this.sliderHandle.addEventListener('mousedown', iniciar);
    this.sliderHandle.addEventListener('touchstart', iniciar);
    document.addEventListener('mouseup', terminar);
    document.addEventListener('touchend', terminar);
    document.addEventListener('mousemove', mover);
    document.addEventListener('touchmove', mover);

    // También mover al hacer clic en la línea
    this.sliderLine.addEventListener('click', (e) => {
      const rect = this.container.getBoundingClientRect();
      const x = e.clientX - rect.left;
      const pct = Math.max(0, Math.min(1, x / rect.width));
      this.setSlider(pct);
    });
  }

  setSlider(pct) {
    this.slider = pct;

    // Clip path para el canvas A (izquierda del slider)
    this.canvasA.style.clipPath = `inset(0 ${100 - pct * 100}% 0 0)`;
    // Clip path para el canvas B (derecha del slider)
    this.canvasB.style.clipPath = `inset(0 0 0 ${pct * 100}%)`;

    // Mover línea y handle
    this.sliderLine.style.left = `${pct * 100}%`;
    this.sliderHandle.style.left = `${pct * 100}%`;
  }

  setModelos(viewerA, viewerB) {
    this.viewerA = viewerA;
    this.viewerB = viewerB;

    // Mover canvas de A al div A
    this.canvasA.innerHTML = '';
    this.canvasA.appendChild(viewerA.renderer.domElement);

    this.canvasB.innerHTML = '';
    this.canvasB.appendChild(viewerB.renderer.domElement);
  }

  setLabels(labelA, labelB) {
    this.container.querySelector('#labelA').textContent = labelA;
    this.container.querySelector('#labelB').textContent = labelB;
  }

  sincronizarCamaras() {
    // Copiar cámara de A a B y viceversa
    const syncAB = () => {
      this.viewerB.camara.position.copy(this.viewerA.camara.position);
      this.viewerB.controles.target.copy(this.viewerA.controles.target);
      this.viewerB.controles.update();
    };

    const syncBA = () => {
      this.viewerA.camara.position.copy(this.viewerB.camara.position);
      this.viewerA.controles.target.copy(this.viewerB.controles.target);
      this.viewerA.controles.update();
    };

    this.viewerA.controles.addEventListener('change', syncAB);
    this.viewerB.controles.addEventListener('change', syncBA);
  }
}

/**
 * ARKI — Time-lapse de evolución del proyecto.
 * Reproduce y graba la evolución de versiones.
 */

export class TimeLapseARKI {
  constructor(viewer, options = {}) {
    this.viewer = viewer;
    this.fps = options.fps || 10;
    this.frames = [];
    this.grabando = false;
  }

  /**
   * Carga los frames desde el historial de versiones.
   */
  async cargarDesdeHistorial(proyectoId) {
    const token = localStorage.getItem('arki_token');
    const res = await fetch(
      `http://localhost:8000/persistencia/versiones/${proyectoId}`,
      { headers: { 'Authorization': `Bearer ${token}` } },
    );

    const data = await res.json();
    const versiones = (data.versiones || []).reverse(); // cronológico

    this.frames = versiones.map(v => ({
      timestamp: v.creado,
      etiqueta: v.etiqueta,
      autor: v.autor_nombre,
      familia_id: v.familia_id,
      variante_id: v.variante_id,
      params: v.params,
      geometria: v.geometria_resumen,
    }));

    return this.frames.length;
  }

  /**
   * Reproduce el time-lapse en el visor.
   */
  async reproducir(callbackProgreso) {
    if (this.frames.length === 0) {
      throw new Error('Sin frames para reproducir');
    }

    for (let i = 0; i < this.frames.length; i++) {
      const frame = this.frames[i];

      // Regenerar geometría desde params
      try {
        const res = await fetch('http://localhost:8000/geometria-especifica/generar', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${localStorage.getItem('arki_token')}`,
          },
          body: JSON.stringify({
            tipo: `${frame.familia_id}_completo`,
            params: frame.params,
          }),
        });

        if (res.ok) {
          const datos = await res.json();
          if (datos.modelo_3d) {
            this.viewer.renderDesdeModelo3D(datos.modelo_3d);
          }
        }
      } catch (e) {
        console.warn(`Frame ${i} falló:`, e);
      }

      if (callbackProgreso) {
        callbackProgreso(i + 1, this.frames.length, frame);
      }

      await new Promise(r => setTimeout(r, 1000 / this.fps));
    }
  }

  /**
   * Graba el time-lapse como video.
   */
  async grabar(nombre = 'timelapse') {
    if (this.frames.length === 0) throw new Error('Sin frames');

    const canvas = this.viewer.renderer.domElement;

    // Canvas virtual para Android
    const virtualCanvas = document.createElement('canvas');
    virtualCanvas.width = canvas.width;
    virtualCanvas.height = canvas.height;
    const vctx = virtualCanvas.getContext('2d');

    const stream = virtualCanvas.captureStream(this.fps);
    const recorder = new MediaRecorder(stream, {
      mimeType: 'video/webm;codecs=vp9',
      videoBitsPerSecond: 25_000_000,
    });

    const chunks = [];
    recorder.ondataavailable = (e) => chunks.push(e.data);
    recorder.start();
    this.grabando = true;

    // Reproducir y capturar
    for (const frame of this.frames) {
      try {
        const res = await fetch('http://localhost:8000/geometria-especifica/generar', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${localStorage.getItem('arki_token')}`,
          },
          body: JSON.stringify({
            tipo: `${frame.familia_id}_completo`,
            params: frame.params,
          }),
        });

        if (res.ok) {
          const datos = await res.json();
          if (datos.modelo_3d) {
            this.viewer.renderDesdeModelo3D(datos.modelo_3d);
          }
        }
      } catch (e) {
        console.warn(e);
      }

      // Render y copiar
      this.viewer.renderer.render(this.viewer.escena, this.viewer.camara);
      vctx.drawImage(canvas, 0, 0);

      await new Promise(r => setTimeout(r, 1000 / this.fps));
    }

    recorder.stop();
    await new Promise(r => { recorder.onstop = r; });
    this.grabando = false;

    const blob = new Blob(chunks, { type: 'video/webm' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${nombre}_${Date.now()}.webm`;
    a.click();
    URL.revokeObjectURL(url);

    return blob;
  }

  getFrames() {
    return this.frames;
  }
}

Notas técnicas críticas
Path tracing requiere WebGL2 y MeshStandardMaterial. La librería no soporta luces puntuales sin MIS activado.

WebXR solo funciona en HTTPS en producción, localhost funciona en desarrollo.

Render nube necesita GPU real (no funciona sin X server ni DISPLAY forwarding en Docker vanilla). Usar nvidia-docker y nvidia-utils.

DOF + Motion Blur son costosos: activar solo cuando se esté quieto.

MediaRecorder tiene comportamiento distinto en Android (necesita canvas virtual). El workaround está implementado.

USDZExporter requiere polyfill de ProgressEvent en Node.js. En navegador funciona directo.

Slider side-by-side requiere dos viewers con la misma cámara. El código incluye sincronización.

Time-lapse consume muchas llamadas al backend. Cachear geometrías si son muchas versiones.
